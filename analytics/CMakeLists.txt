if(CONFIG_ZMK_ANALYTICS)
  if(CONFIG_LTO OR CMAKE_INTERPROCEDURAL_OPTIMIZATION OR
     "${CMAKE_C_FLAGS} ${EXTRA_CFLAGS}" MATCHES "-flto")
    message(FATAL_ERROR "Analytics PoC requires LTO disabled for linker wrapping")
  endif()
  # Refuse unreviewed source changes instead of silently losing the exactly-once property.
  include(${CMAKE_CURRENT_LIST_DIR}/cmake/audited-sources.cmake)
  zephyr_library()
  zephyr_library_include_directories(include ${APPLICATION_SOURCE_DIR}/include)
  zephyr_library_sources(src/collector.c src/position_hook.c src/storage.c)
  zephyr_library_compile_definitions(ANALYTICS_ZMK_BUILD=1)
  zephyr_link_libraries(-Wl,--wrap=raise_zmk_position_state_changed)
  zephyr_library_sources_ifdef(CONFIG_ZMK_ANALYTICS_USB_POC src/usb_poc.c)
  if(CONFIG_ZMK_ANALYTICS_PERSISTENCE)
    if("${CONFIG_ZMK_ANALYTICS_LAYOUT_ID}" STREQUAL "")
      message(FATAL_ERROR "Analytics persistence requires a versioned physical layout ID")
    endif()
    if(NOT "${CONFIG_ZMK_ANALYTICS_LAYOUT_ID}" MATCHES "^[A-Za-z0-9:._-]+$")
      message(FATAL_ERROR "Analytics layout ID may contain only A-Z, a-z, 0-9, colon, dot, underscore, hyphen")
    endif()
    string(SHA256 layout_hash "${CONFIG_ZMK_ANALYTICS_LAYOUT_ID}")
    string(REGEX REPLACE "(..)" "0x\\1," layout_hash_bytes "${layout_hash}")
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/analytics_layout_hash.h"
      "static const uint8_t analytics_layout_hash[32] = {${layout_hash_bytes}};\n")
    zephyr_library_include_directories(${CMAKE_CURRENT_BINARY_DIR})
    zephyr_library_sources(src/record.c)
  endif()
  if(CONFIG_ZMK_ANALYTICS_POC_REPORT)
    zephyr_library_sources(tests/native_report.c)
  endif()
endif()
