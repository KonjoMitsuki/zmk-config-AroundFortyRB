cmake_minimum_required(VERSION 3.20)
project(analytics_collector_tests C)
enable_testing()
option(ANALYTICS_TEST_SANITIZERS "Check host tests for memory errors and undefined behavior" OFF)
if(ANALYTICS_TEST_SANITIZERS)
  add_compile_options(-fsanitize=address,undefined -fno-sanitize-recover=all
                      -fno-omit-frame-pointer -g)
  add_link_options(-fsanitize=address,undefined)
endif()
add_executable(test_collector test_collector.c ../src/collector.c)
target_include_directories(test_collector PRIVATE ../include)
target_compile_definitions(test_collector PRIVATE ANALYTICS_POSITION_COUNT=42)
target_compile_options(test_collector PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME collector COMMAND test_collector)
add_executable(test_record test_record.c ../src/record.c ../src/collector.c ../src/bigram.c)
target_include_directories(test_record PRIVATE ../include)
target_compile_definitions(test_record PRIVATE ANALYTICS_POSITION_COUNT=42)
target_compile_options(test_record PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME record COMMAND test_record)
add_executable(test_timing test_timing.c ../src/timing.c ../src/collector.c ../src/bigram.c
               ../src/recent.c)
target_include_directories(test_timing PRIVATE ../include)
target_compile_definitions(test_timing PRIVATE ANALYTICS_POSITION_COUNT=42)
target_compile_options(test_timing PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME timing COMMAND test_timing)
